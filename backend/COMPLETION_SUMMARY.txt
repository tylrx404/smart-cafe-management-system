## 🔌 CONNECTING REACT

In your React components:

```javascript
const BASE_URL = 'http://localhost:5000/api';

// Create order
const response = await fetch(`${BASE_URL}/orders`, {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    table_id: 1,
    items: [{name: 'Tea', price: 50, quantity: 1}],
    total_amount: 50
  })
});

// Fetch orders
const orders = await fetch(`${BASE_URL}/orders`).then(r => r.json());

// Update status
await fetch(`${BASE_URL}/orders/1/status`, {
  method: 'PUT',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ status: 'preparing' })
});

// Record payment
await fetch(`${BASE_URL}/payments`, {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    order_id: 1,
    amount: 50,
    payment_method: 'cash',
    payment_status: 'completed'
  })
});
```

See **API_EXAMPLES.js** for more examples!
